% Range grid cell: \rangecell{col}{row}{raise frac}{call frac}{label}. A 13x13 grid has
% col, row = 0..12 (0 = ace), pairs on the diagonal, suited above it. Fractions in [0,1];
% the cell is filled left to right raise | call | fold.
\newcommand{\rangecell}[5]{%
  \fill[foldcol] (#1,-#2) rectangle ++(1,-1);
  \fill[raisecol] (#1,-#2) rectangle ++(#3,-1);
  \fill[callcol] ({#1+#3},-#2) rectangle ++(#4,-1);
  \draw[white,line width=0.3pt] (#1,-#2) rectangle ++(1,-1);
  \node[font=\fontsize{4}{4}\selectfont,inner sep=0] at ({#1+0.5},{-#2-0.5}) {#5};}
